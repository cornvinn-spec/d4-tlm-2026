  <script>
    // Fade-in on scroll
    const revealObserver = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add('visible');
          revealObserver.unobserve(entry.target);
        }
      });
    }, { threshold: 0.15 });
    document.querySelectorAll('.reveal').forEach(el => revealObserver.observe(el));

    // Generate member grid
    const memberGrid = document.getElementById('memberGrid');
    for (let i = 1; i <= 24; i++) {
      const card = document.createElement('div');
      card.className = 'reveal glass rounded-xl p-4 text-center hover:shadow-neon hover:-translate-y-1 transition';
      card.innerHTML = `
        <img src="https://ui-avatars.com/api/?name=Anggota+${i}&background=0f172a&color=00f0ff&size=80"
             alt="Anggota ${i}" class="w-16 h-16 rounded-full mx-auto mb-3">
        <p class="text-sm font-medium text-white">Nama Anggota ${i}</p>
        <p class="text-xs text-slate-400">NIM: 00000${i}</p>`;
      memberGrid.appendChild(card);
      revealObserver.observe(card);
    }

    // Carousel
    const slides = document.getElementById('slides');
    const total = slides.children.length;
    const dotsWrap = document.getElementById('dots');
    let current = 0;

    for (let i = 0; i < total; i++) {
      const dot = document.createElement('button');
      dot.className = 'w-2.5 h-2.5 rounded-full bg-slate-600 transition';
      dot.setAttribute('aria-label', `Slide ${i + 1}`);
      dot.addEventListener('click', () => goTo(i));
      dotsWrap.appendChild(dot);
    }

    function goTo(index) {
      current = (index + total) % total;
      slides.style.transform = `translateX(-${current * 100}%)`;
      [...dotsWrap.children].forEach((d, i) => {
        d.className = `w-2.5 h-2.5 rounded-full transition ${i === current ? 'bg-neon shadow-neon' : 'bg-slate-600'}`;
      });
    }

    document.getElementById('prevBtn').addEventListener('click', () => goTo(current - 1));
    document.getElementById('nextBtn').addEventListener('click', () => goTo(current + 1));
    setInterval(() => goTo(current + 1), 5000);
    goTo(0);

    // Static contact form
    document.getElementById('contactForm').addEventListener('submit', (e) => {
      e.preventDefault();
      const msg = document.getElementById('contactMsg');
      msg.textContent = 'Terima kasih! Pesan Anda telah diterima (mode statis).';
      msg.classList.remove('hidden');
      e.target.reset();
    });

    // Static guestbook
    document.getElementById('guestForm').addEventListener('submit', (e) => {
      e.preventDefault();
      const [nameInput, msgInput] = e.target.querySelectorAll('input, textarea');
      const li = document.createElement('li');
      li.className = 'bg-slate-900/50 rounded-xl p-4 border border-slate-800';
      li.innerHTML = `<p class="font-semibold text-white"></p><p class="text-sm text-slate-400"></p>`;
      li.querySelector('p:first-child').textContent = nameInput.value;
      li.querySelector('p:last-child').textContent = msgInput.value;
      document.getElementById('guestList').prepend(li);
      e.target.reset();
    });
  </script>
